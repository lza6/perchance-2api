      if(e.data.type === "rateGeneratedText") {

    async function middleOut(str, maxTokens) {
      // NOTE: If you upgrade the model and the tokenizer adds a BOS token, then you need to return `window.tokenizer.decode(tokens.slice(1))` instead
      let tokens = window.tokenizer.encode(str);
      const len = tokens.length;
      if(len <= maxTokens) return window.tokenizer.decode(tokens);

      const numToRemove = len - maxTokens;
      const startIdx = Math.floor((len - numToRemove) / 2);
      
      tokens.splice(startIdx, numToRemove, ...window.tokenizer.encode("\n\n[...]\n\n"));
      return window.tokenizer.decode(tokens);
    }

    function middleOutWithoutTokenizer(str, maxStringLength) {
      if(str.length <= maxStringLength) return str;

      const numCharsToRemove = str.length - maxStringLength;
      const startIndex = Math.floor((str.length - numCharsToRemove) / 2);

      const arr = [...str];
      arr.splice(startIndex, numCharsToRemove, "\n\n[...]\n\n");

      return arr.join("");
    }


    function moveToLeastRecentlyUsedThread() {
      let leastRecentTime = generationIdentity.storage[`threadLastUsedTime-0`] || 0;
      let leastRecentThreadIndex = 0;
      for(let i = 0; i < maxThreadsPerUser; i++) {
        let lastTimeUsed = generationIdentity.storage[`threadLastUsedTime-${i}`] || 0;
        if(lastTimeUsed < leastRecentTime) {
          leastRecentTime = lastTimeUsed;
          leastRecentThreadIndex = i;
        }
      }
      window.thread = leastRecentThreadIndex;
    }

    function djb2Hash(str) {
      let hash = 5381;
      for (let i = 0; i < str.length; i++) {
        hash = ((hash << 5) + hash) + str.charCodeAt(i); // hash * 33 + c
      }
      return hash >>> 0; // Convert to unsigned 32-bit
    }

    // Mints a single-use ticket that lets the page's server plugin run one generation as this user.
    async function createTicket({kind, generatorName}) {
      if(kind !== "generateText") throw new Error("invalid_params: unsupported ticket kind");
      if(typeof generatorName !== "string" || !generatorName) throw new Error("invalid_params: generatorName is required");
      await generationIdentity.ensure();
      if(generationIdentity.storage[`threadGenerationInProgress-${window.thread}`]) moveToLeastRecentlyUsedThread();
      const thread = window.thread;
      generationIdentity.storage[`threadLastUsedTime-${thread}`] = Date.now(); // the ticket's generation will use this thread
      let lastStatus = null;
      for(let attempt = 0; attempt < 4; attempt++) {
        const userKey = generationIdentity.storage[`userKey-${thread}`] || "";
        const result = await fetch(`/api/createTicket?browserId=${generationIdentity.id}&userKey=${encodeURIComponent(userKey)}&thread=${thread}&__cacheBust=${Math.random()}`, {
          method: "POST",
          body: JSON.stringify({kind, generatorName}),
        }).then(r => r.json()).catch(e => ({status:"network_failure"}));
        lastStatus = result.status;
        if(result.status === "success") return {ticket:result.ticket, expiresAt:result.expiresAt};
        if(result.status === "invalid_key" || result.status === "need_reverification_soon") {
          threadIndexToLastSuccessfulVerificationCheckTime[thread] = 0; // to ensure this doesn't block reverification
          await verifyUser(thread);
        } else if(result.status === "network_failure") {
          await window.hiddenSafeWait(2000 + Math.random()*2000);
        } else if(result.status === "client_update_required" || result.status === "invalid_browser_id") {
          throw new Error("client_update_required: Please refresh the page to continue.");
        } else if(result.status === "too_many_requests" || result.status === "too_many_tickets") {
          throw new Error("quota_exceeded: too many requests, please try again later");
        } else {
          throw new Error(`service_unavailable: ${result.status}`);
        }
      }
      throw new Error(lastStatus === "network_failure" ? "network_failure: couldn't reach the server" : "verification_failed: couldn't verify the user");
    }

    async function startStreaming(postData, requestId, opts={}) {
      await generationIdentity.ensure();
      if (window.requestIdsToStop.has(requestId)) return;
      if(generationIdentity.storage[`threadGenerationInProgress-${window.thread}`]) {
        moveToLeastRecentlyUsedThread(); // note: it's okay if we sometimes switch to a thread that is already being used - it'll just queue up and take a bit longer
      }
      let thread = window.thread;

      // this is a long-running request that will return a response when the previous /api/generate request has finished.
      // but note that there could be multiple requests waiting - but in that case /api/generate will return "waiting_for_prev_request_to_finish", and we'll wait and try again.
      if(opts.shouldAwaitExistingGenerationRequest) {
        let signal = window.AbortSignal && AbortSignal.timeout ? AbortSignal.timeout(1000*20) : null;
        let result = await fetch(`/api/awaitExistingGenerationRequest?userKey=${opts.waitUserKey || generationIdentity.storage[`userKey-${thread}`]}&__cacheBust=${Math.random()}`, {signal}).then(r => r.json()).catch(e => ({status:"fetch_failure"}));
        if(result.status === "invalid_key") {
          threadIndexToLastSuccessfulVerificationCheckTime[thread] = 0; // to ensure this doesn't block reverification
          await window.hiddenSafeWait(1000);
          let success = await verifyUser(thread);
          if(success) {
            window.parent.postMessage({type:"verified"}, window.perchanceGeneratorOrigin);
          } else {
            // it's okay to not handle this because startStreaming loops if /api/generate returns invalid_key
          }
        } else if(result.status === "verified") {
          window.parent.postMessage({type:"verified"}, window.perchanceGeneratorOrigin);
        }
        opts.shouldAwaitExistingGenerationRequest = false;
      }

      // should probably be longer than 1 second since I *think* backgrounded tabs may slow down timers (e.g. setTimeout, setInterval) to 1 per second after a bit?
      // edit: this needs to be at least 5 seconds due to random browser lag (and potentially bugs which I've yet to track down RE slow 'keepalive' messages)
      // increasing yet again to 10. we have requestIdsToStop now so it should be fine. probably not a lag issue that this point - maybe keepalive requests getting lost somehow, or stopping too early.
      const requestStalenessTimeout = 10000;

      if(Date.now() - requestIdToLastStreamKeepAliveTime[requestId] > requestStalenessTimeout) {
        console.warn("Stream keep alive timeout. Ignoring this stale request.");
        window.requestIdToDidTimeOut[requestId] = true;
        window.consecutiveStrangeFailCount++;

        // debug
        try {
          if(!window.recentDebugEvents) { window.recentDebugEvents = []; };
          if(window.recentDebugEvents.length > 20) window.recentDebugEvents = window.recentDebugEvents.slice(-20);

        } catch(e) { console.error(e); }

        // try { rollbar.critical(`stale1 ${requestId} ${Date.now()}`); } catch(e) { console.error(e); }

      }
      generationIdentity.storage.lastGenerateRequestTime = Date.now();

      generationIdentity.storage[`threadLastUsedTime-${thread}`] = Date.now();
      generationIdentity.storage[`threadGenerationInProgress-${thread}`] = "1"; // <-- note: this doesn't actually have to be full-proof (e.g. it's fine if they close the page before we set this back to a falsy value), it's just to help us choose the best thread to use
      
      let generatedText = ""; // just for debug

      let handles = {};

          } catch(e) { console.error(e); }

          if(window[`recentAbortedRequestIds-${thread}`].length > 10) window[`recentAbortedRequestIds-${thread}`] = window[`recentAbortedRequestIds-${thread}`].slice(-10);
          window[`recentAbortedRequestIds-${thread}`].push(requestId);

      try {

        if(!window.lastRequestIdByThread) window.lastRequestIdByThread = {};

        // debug
        if(!window.lastPromptHashesByThread) window.lastPromptHashesByThread = {};
        window.lastPromptHashesByThread[thread] = {instructionHash:djb2Hash(postData.instruction+""), startWithHash:djb2Hash(postData.startWith+"")};

    async function* eventStreamToAsyncIterator(url, postData, handles={}, opts={}) {

      let started = false;
      let finished = false;
      let connectionTimeout;
      let connectionTimeoutActive = true;
      try {
        const controller = new AbortController(); 
        const signal = controller.signal;
        handles.controller = controller; // this gives the caller of this function the ability to abort this stream

        let connectionTimeoutLength = 1000*60;
        if(window.thereWasARecentNetworkErrorWhileStreaming) connectionTimeoutLength = 1000*90; // may indicate server restart, so need to give it a bit more time.

        connectionTimeout = setTimeout(async () => {
          if(!connectionTimeoutActive || started || finished || signal.aborted) return;
          
          let { stillWorking } = await fetch(`/api/checkRequestId?requestId=${opts.requestId}&userKey=${opts.userKey}`).then(r => r.json()).catch(e => ({stillWorking:false}));
          if(!connectionTimeoutActive || started || finished || signal.aborted) return; // the status query may finish after headers
          if(stillWorking) await window.hiddenSafeWait(1000*60); // maybe under heavy load, so give it more time
          
          if(!connectionTimeoutActive || started || finished || signal.aborted) return; // status fetch may outlive the pre-header wait
          console.warn("ABORTING DUE TO CONNECTION TIMEOUT", connectionTimeoutLength);
          controller.abort();
        }, connectionTimeoutLength);

        // if aborted by parent function via `handles`, the clear timeout:
        signal.addEventListener("abort", () => {
          connectionTimeoutActive = false;
          clearTimeout(connectionTimeout);
        });

        // Body inactivity starts at response headers, even if no body byte has
        // arrived. The server withholds its old blank-line first byte so it can
        // retry an unstreamed Cloudflare error; headers alone must still arm a
        // timeout. Each body chunk resets it, preserving long active streams.
        // hiddenSafeWait handles hidden-iframe timer throttling.
        var lastStreamActivityTime = Date.now();
        let bodyWatchdogArmed = false;
        (async () => {
          while(!finished && !signal.aborted) {
            await window.hiddenSafeWait(5000);
            if(finished || signal.aborted) break;
            if(bodyWatchdogArmed && Date.now() - lastStreamActivityTime > 60000) {
              console.warn("ABORTING DUE TO RESPONSE BODY STALL (no bytes for 60s)");
              controller.abort();
              break;
            }
          }
        })();

        const response = await fetch(url, {
          signal,
          method: "POST",
          body: JSON.stringify(postData),
        });
        connectionTimeoutActive = false;
        clearTimeout(connectionTimeout);
        lastStreamActivityTime = Date.now();
        bodyWatchdogArmed = true;

        if(response.headers.get("x-new-user-key") && opts.thread !== undefined) {
          const newKey = response.headers.get("x-new-user-key");
          opts.userKey = newKey; // This request owns the key even if the shared cache moved on.
          generationIdentity.setKey(opts.thread, newKey, opts.identityId);
        }
        if(response.headers.get("x-should-reverify") === "1") {
          console.debug("x-should-reverify header found. re-verifying user in the background while generating...");
          (async function() {
            let result = await verifyUser(0, {lowPriority:true});
            if(result) {
              console.debug("Re-verified user.");
            } else {
              console.error("Failed to re-verify user.");
            }
          })();
        }

        if(response.status !== 200) {
          const json = await response.json().catch(e => ({status:`${response.status}`}));
          yield { error:true, ...json };
          return;
        }

        if(window.TextDecoderStream) {
          const textStream = response.body.pipeThrough(new window.TextDecoderStream());
          const reader = textStream.getReader();
          let buffer = "";

          while(true) {
            const { done, value } = await reader.read();
            lastStreamActivityTime = Date.now();
            if(done) break;
            started = true;

            buffer += value;
            let lineEndIndex;

            while((lineEndIndex = buffer.indexOf('\n')) >= 0) {
              const line = buffer.slice(0, lineEndIndex).trim();
              buffer = buffer.slice(lineEndIndex + 1);

              if(line.startsWith('t:')) { // custom, more-efficient protocol
                const text = JSON.parse(line.slice(2));
                yield {text};
              } else if(line.startsWith('data:')) {
                const jsonData = JSON.parse(line.slice(5).trim());
                yield jsonData;
              } 
            }
          }
        } else {
          // for Safari 14:
          const reader = response.body.getReader();
          let buffer = "";
          let textDecoder = new TextDecoder();

          while(true) {
            const { done, value } = await reader.read();
            lastStreamActivityTime = Date.now();
            if(done) break;
            started = true;

            const chunk = textDecoder.decode(value, { stream: true });
            buffer += chunk;

            let lineEndIndex;
            while((lineEndIndex = buffer.indexOf('\n')) >= 0) {
              const line = buffer.slice(0, lineEndIndex).trim();
              buffer = buffer.slice(lineEndIndex + 1);

              if(line.startsWith('t:')) { // custom, more-efficient protocol
                const text = JSON.parse(line.slice(2));
                yield {text};
              } else if(line.startsWith('data:')) {
                const jsonData = JSON.parse(line.slice(5).trim());
                yield jsonData;
              }
            }
          }
        }
      } finally {
        finished = true;
        connectionTimeoutActive = false;
        clearTimeout(connectionTimeout);
      }
    }

              }, 1000*4);

                try { fetch(`/api/verificationDebug?status=error-callback&args=${encodeURIComponent(JSON.stringify(args))}`); } catch(e) { console.error(e); }
              },

            let waitLogInterval = setInterval(() => {

              window.lastActivelyVerifyingTime = Date.now();
              console.debug("Waiting for token...", waitedSeconds, ctnString.slice(0, 10));
              waitedSeconds++;
              if(waitedSeconds === 5 && ctnString === "") { // I think cloudflare (sometimes??) doesn't render the widget if a separate verification is already in progress.
                window["cloudflareTurnstileTokenResolver-"+thread](null);
              }
              if(waitedSeconds > 60) { // sometimes cloudflare can 'hang' indefinitely for some reason, so in that case we just cancel it
                window["cloudflareTurnstileTokenResolver-"+thread](null);

                  window.__sentTurnstileTimeoutBeacon = true;
                  try { fetch(`/api/verificationDebug?status=client-timeout`); } catch(e) {}
                }
                window.consecutiveTurnstileTimeouts++;
                if(window.consecutiveTurnstileTimeouts >= 2) {
                  window.location.reload();
                }
              }
            }, 1000);
            
            let token = await tokenPromise.catch(e => { console.error(e); return null; });

            if(token !== null) window.consecutiveTurnstileTimeouts = 0;
            
            clearInterval(waitLogInterval);
            clearTimeout(window.verifyingPostMessageTimeout);
            try { window.turnstile.remove(window.turnstileWidgetId); } catch(e) { console.error("turnstile.remove() failed:", e); }

            if(token === null) {
              await window.hiddenSafeWait(3000); // pause for a bit if we failed - to avoid spamming cloudflare
              resolve(false);
              return;
            }

            let fetchOptions = {};
            if(window.AbortSignal && AbortSignal.timeout) fetchOptions.signal = AbortSignal.timeout(1000*15);

            result = await fetch(`/api/verifyUser?browserId=${generationIdentity.id}&token=${encodeURIComponent(token)}&thread=${thread}&__cacheBust=${Math.random()}`, fetchOptions).then(r => r.json()).catch(e => ({status:"fetch_failure"}));
          }

          clearTimeout(window.verifyingPostMessageTimeout);
          
          if(result.status === "success" || result.status === "already_verified") {
            if (!generationIdentity.setKey(thread, result.userKey, verificationIdentityId)) { resolve(false); return; }
            console.debug("Verification succeeded.");
            resolve(generationIdentity.storage[`userKey-${thread}`]);
            window.parent.postMessage({type:"verified"}, window.perchanceGeneratorOrigin);
            if(result.attemptedFullVerification) fetch(`/api/verificationDebug?status=success`);
          } else {
            if(result.attemptedFullVerification) fetch(`/api/verificationDebug?status=fail`);
            console.debug("Turnstile verification failed.", result);
            resolve(false);
          }
        } catch(e) {
          clearTimeout(window.verifyingPostMessageTimeout);
          console.error("caught verification error of some kind:", e);
          resolve(false);
          fetch(`/api/verificationDebug?status=fail`);
        }
      });

      let result = await window.threadIndexToCurrentlyVerifyingUserPromise[thread].catch(e => { console.error(e); return false; });
      window.threadIndexToCurrentlyVerifyingUserPromise[thread] = null;

      return result;
    }
